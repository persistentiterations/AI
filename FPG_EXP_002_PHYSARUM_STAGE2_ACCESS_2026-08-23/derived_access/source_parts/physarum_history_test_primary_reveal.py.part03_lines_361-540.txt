    np.savez_compressed(output, **arrays)


def frame_equal_weights(frames: np.ndarray) -> np.ndarray:
    unique, counts = np.unique(frames, return_counts=True)
    mapping = dict(zip(unique.tolist(), counts.tolist()))
    return np.asarray([1.0 / mapping[int(f)] for f in frames], dtype=np.float64)


def standardize_train_test(x_train: np.ndarray, x_test: np.ndarray, w: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    mean = np.average(x_train, axis=0, weights=w)
    var = np.average((x_train - mean) ** 2, axis=0, weights=w)
    scale = np.sqrt(np.maximum(var, 1e-8))
    return (x_train - mean) / scale, (x_test - mean) / scale


def weighted_log_loss(y: np.ndarray, p: np.ndarray, w: np.ndarray) -> float:
    p = np.clip(p, 1e-7, 1 - 1e-7)
    return float(np.average(-(y * np.log(p) + (1 - y) * np.log(1 - p)), weights=w))


def quantile_bins(values: np.ndarray, count: int) -> np.ndarray:
    order = np.argsort(values, kind="stable")
    bins = np.empty(len(values), dtype=np.int16)
    bins[order] = np.minimum(count - 1, (count * np.arange(len(values))) // max(1, len(values)))
    return bins


def matched_shuffle_history(
    history: np.ndarray,
    frames: np.ndarray,
    snapshot: np.ndarray,
    snapshot_p: np.ndarray,
    seed: int,
) -> np.ndarray:
    """Shuffle rows within time, arena, and snapshot-propensity strata."""
    time_bin = quantile_bins(frames.astype(np.float64), 5)
    annulus_bin = quantile_bins(snapshot[:, 17], 4)
    propensity_bin = quantile_bins(snapshot_p, 10)
    key = time_bin.astype(np.int32) * 100 + annulus_bin.astype(np.int32) * 10 + propensity_bin.astype(np.int32)
    out = history.copy()
    rng = np.random.default_rng(seed)
    for value in np.unique(key):
        idx = np.flatnonzero(key == value)
        if len(idx) > 1:
            out[idx] = history[rng.permutation(idx)]
    return out


def stratum_deltas(y: np.ndarray, ps: np.ndarray, ph: np.ndarray, weights: np.ndarray, values: np.ndarray) -> list[float]:
    bins = quantile_bins(values, 4)
    out = []
    for b in range(4):
        take = bins == b
        out.append(weighted_log_loss(y[take], ps[take], weights[take]) - weighted_log_loss(y[take], ph[take], weights[take]))
    return out
